# PART 1: SYSTEM ARCHITECTURE & DESIGN GOALS

The clj-xla project is a pure Clojure library and runtime binding for OpenXLA PJRT. It provides direct C-FFI bindings to PJRT plugin dynamic libraries (such as libpjrt_cpu.so, libpjrt_rocm.so, and libpjrt_cuda.so) via Java 22+ Foreign Function & Memory API (JEP 454).

## 1.1 Core Principles & Operating Constraints
- Rule 1: Strict TDD. Write generative property tests using clojure.test.check for invariants before implementing core logic.
- Rule 2: Pure Functions. The core graph builder and StableHLO compiler must remain pure Sans-IO functions. Side effects are strictly isolated to boundary shells.
- Rule 3: Clean CI. Pull requests must pass formatting, linter, and unit/generative test runners without warnings or failures.
- Rule 4: Zero Java Escape Hatches (Pure XLA Execution). All tensor math, neural network layers, KV-cache updates, attention mechanisms, and full model forward passes MUST be written in pure Clojure using clj-xla.tensor and clj-xla.trace and compiled into StableHLO MLIR for OpenXLA execution via PJRT. Under no circumstances should custom .java classes, host-side primitive float array loops, or manual CPU matrix math engines be created to bypass XLA compilation.

## 1.2 Module Overview

1. clj-xla.core: High-level session management, device buffer allocation, XLA flag autotuning, and execution graph compilation.
2. clj-xla.pjrt: Direct C-FFI wrapper around PJRT C API function pointers using Java Foreign Function & Memory (FFM) API downcall handles.
3. clj-xla.stablehlo: Intermediate Representation (IR) emitter that serializes EDN SSA graph tracing records into text-format StableHLO MLIR.
4. clj-xla.tensor: Functional tracer API defining basic linear algebra, math, slice, reshape, transpose, gather, compare, and reduction operators.
5. clj-xla.trace: Execution tracer context atom that records operator equations (SSA bindings) during function execution.
6. clj-xla.nn.attention: Multi-head attention (MHA), grouped-query attention (GQA), RoPE positional embeddings, and sliding-window KV-cache updating algorithms.
7. clj-xla.models.gemma: Architecture specifications, hyperparameter definitions, per-layer weight mapping, and monolithic forward execution graphs for Google Gemma 4 models (E2B, 12B, 26B, 31B).

(ns clj-xla.tensor
  "Pure functional tensor tracer operations for StableHLO graph construction."
  (:refer-clojure :exclude [+ - * / max min count get abs exp log sin cos tan sinh cosh tanh reshape concat compare])
  (:require [clj-xla.trace :refer [*trace-ctx* gen-var-id!]]))

(defrecord Tracer [id type])

(defn emit-constant! [val target-type]
  (cond
    (instance? Tracer val) val
    (number? val)
    (let [out-id (gen-var-id! "c_num")
          out-type (or target-type [:tensor [] :f32])
          eqn {:op :stablehlo/constant :outvars [out-id] :value val :type out-type}]
      (when *trace-ctx*
        (swap! (:eqns *trace-ctx*) conj eqn))
      (->Tracer out-id out-type))
    (vector? val)
    (let [out-id (gen-var-id! "c_vec")
          out-type (or target-type [:tensor [(clojure.core/count val)] :f32])
          eqn {:op :stablehlo/constant :outvars [out-id] :value val :type out-type}]
      (when *trace-ctx*
        (swap! (:eqns *trace-ctx*) conj eqn))
      (->Tracer out-id out-type))))

(defn- broadcast-shapes [shape-a shape-b]
  (let [rank-a (clojure.core/count shape-a)
        rank-b (clojure.core/count shape-b)
        max-rank (clojure.core/max rank-a rank-b)
        pad-a (vec (clojure.core/concat (repeat (clojure.core/- max-rank rank-a) 1) shape-a))
        pad-b (vec (clojure.core/concat (repeat (clojure.core/- max-rank rank-b) 1) shape-b))]
    (mapv clojure.core/max pad-a pad-b)))

(defn- binary-elementwise-op
  ([op-kw prefix a b] (binary-elementwise-op op-kw prefix a b nil nil))
  ([op-kw prefix a b attrs out-dtype-override]
   (let [ta (emit-constant! a nil)
         tb (emit-constant! b (:type ta))
         [t-kw shape-a dtype-a] (:type ta)
         [_ shape-b _] (:type tb)
         out-shape (broadcast-shapes shape-a shape-b)
         rank-out (clojure.core/count out-shape)
         ta-bcast (if (= shape-a out-shape)
                    ta
                    (let [rank-a (clojure.core/count shape-a)
                          bcast-dims (vec (map int (range (clojure.core/- rank-out rank-a) rank-out)))]
                      (broadcast-in-dim ta out-shape bcast-dims)))
         tb-bcast (if (= shape-b out-shape)
                    tb
                    (let [rank-b (clojure.core/count shape-b)
                          bcast-dims (vec (map int (range (clojure.core/- rank-out rank-b) rank-out)))]
                      (broadcast-in-dim tb out-shape bcast-dims)))
         out-type [t-kw out-shape (or out-dtype-override dtype-a)]
         out-id (gen-var-id! prefix)
         eqn (cond-> {:op op-kw :invars [(:id ta-bcast) (:id tb-bcast)] :outvars [out-id]}
               attrs (assoc :attrs attrs))]
     (when *trace-ctx*
       (swap! (:eqns *trace-ctx*) conj eqn))
     (->Tracer out-id out-type))))

(defn + [a b] (binary-elementwise-op :stablehlo/add "t_add" a b))
(defn - [a b] (binary-elementwise-op :stablehlo/subtract "t_sub" a b))
(defn * [a b] (binary-elementwise-op :stablehlo/multiply "t_mul" a b))
(defn / [a b] (binary-elementwise-op :stablehlo/divide "t_div" a b))
(defn compare-t [a b direction] (binary-elementwise-op :stablehlo/compare "t_cmp" a b {:comparison_direction (str direction)} :i1))
(defn eq [a b] (compare-t a b "EQ"))

(defn argmax
  [x & {:keys [axis] :or {axis -1}}]
  (let [tx (emit-constant! x nil)
        [_kw shape _dtype] (:type tx)
        rank (clojure.core/count shape)
        norm-axis (if (clojure.core/neg? axis) (clojure.core/+ rank axis) axis)
        dim-size (nth shape norm-axis)
        max-val (reduce-max tx :axes [norm-axis] :keep-dims true)
        is-max (eq tx max-val)
        idx-shape (assoc (vec (repeat rank 1)) norm-axis dim-size)
        idx-const (emit-constant! (vec (range dim-size)) [:tensor idx-shape :i32])
        masked-idx (* (convert is-max :i32) idx-const)]
    (reduce-max masked-idx :axes [norm-axis] :keep-dims false)))

```clojure
(defn dot-general
  [lhs rhs & {:keys [contracting-dims batch-dims] :or {contracting-dims {:lhs [1] :rhs [0]} batch-dims {:lhs [] :rhs []}}}]
  (let [tlhs (emit-constant! lhs nil)
        trhs (emit-constant! rhs nil)
        [t-kw shape-lhs dtype] (:type tlhs)
        [_ shape-rhs _] (:type trhs)
        c-lhs (get contracting-dims :lhs [1])
        c-rhs (get contracting-dims :rhs [0])
        b-lhs (get batch-dims :lhs [])
        b-rhs (get batch-dims :rhs [])
        b-dims (mapv #(nth shape-lhs %) b-lhs)
        c-lhs-set (set c-lhs)
        c-rhs-set (set c-rhs)
        b-lhs-set (set b-lhs)
        b-rhs-set (set b-rhs)
        lhs-free (keep-indexed (fn [idx dim] (when-not (or (c-lhs-set idx) (b-lhs-set idx)) dim)) shape-lhs)
        rhs-free (keep-indexed (fn [idx dim] (when-not (or (c-rhs-set idx) (b-rhs-set idx)) dim)) shape-rhs)
        out-shape (vec (concat b-dims lhs-free rhs-free))
        out-type [t-kw out-shape dtype]
        out-id (gen-var-id! "t_dot")
        eqn {:op :stablehlo/dot_general
             :invars [(:id tlhs) (:id trhs)]
             :outvars [out-id]
             :attrs {:contracting_dims {:lhs c-lhs :rhs c-rhs}
                     :batch_dims {:lhs b-lhs :rhs b-rhs}}}]
    (when *trace-ctx*
      (swap! (:eqns *trace-ctx*) conj eqn))
    (->Tracer out-id out-type)))
```

  (let [b 1
        q-seq-len (second (second (:type q)))
        kv-seq-len max-seq-len

        k-rot (apply-rotary-embedding k position-idx head-dim rope-proportion)
        [updated-k updated-v] (update-kv-cache kv-cache k-rot v position-idx)

        k-expanded (repeat-kv-heads updated-k group-size)
        v-expanded (repeat-kv-heads updated-v group-size)
        scale (/ 1.0 (Math/sqrt head-dim))

        scaled-scores (* scores scale)
        attn-weights (softmax scaled-scores :axis -1)
        context (dot-general attn-weights v-expanded :contracting-dims {:lhs [3] :rhs [2]} :batch-dims {:lhs [0 1] :rhs [0 1]})]
    [context updated-k updated-v]))

(defn argmax
  [x & {:keys [axis] :or {axis -1}}]
  (let [tx (emit-constant! x nil)
        [_kw shape _dtype] (:type tx)
        rank (clojure.core/count shape)
        norm-axis (if (clojure.core/neg? axis) (clojure.core/+ rank axis) axis)
        dim-size (nth shape norm-axis)
        max-val (reduce-max tx :axes [norm-axis] :keep-dims true)
        is-max (eq tx max-val)
        idx-shape (assoc (vec (repeat rank 1)) norm-axis dim-size)
        idx-const (emit-constant! (vec (range dim-size)) [:tensor idx-shape :i32])
        masked-idx (* (convert is-max :i32) idx-const)]
    (reduce-max masked-idx :axes [norm-axis] :keep-dims false)))
